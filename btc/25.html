EC_KEY,是给原始的ECDS_sign使用的。CKey对它们进行了封装，接下来我们脱离CKey来使用这些函数，这样我们就可以给随意长度的数据进行签名和验证了。

注意：不推荐这种，只是为了研究学习使用，签名还是先将数据生成哈希值再进行，一个是安全性提高，二个，如果数据非常大，这样直接签名会很呆。

8.openssl签名与验证

直接看例子吧，原始用法，任意长度数据，非哈希值：

  1. // 打印 hex 辅助函数
  2. void PrintHex(const char* name, const unsigned char* data, size_t len)
  3. {
  4.     std::cout << name << ": ";
  5.     for (size_t i = 0; i < len; ++i)
  6.         std::cout << std::hex << std::setw(2) << std::setfill('0') << (int)data[i];
  7.     std::cout << std::dec << std::endl;
  8. }
  9.  
  10. int main()
  11. {
  12.     std::cout << "=== OpenSSL ECDSA 直接签名示例（不计算哈希）===\n" << std::endl;
  13.  
  14.     // 1. 创建 secp256k1 密钥
  15.     EC_KEY* eckey = EC_KEY_new_by_curve_name(NID_secp256k1);
  16.     if (!EC_KEY_generate_key(eckey)) {
  17.         std::cerr << "生成密钥失败" << std::endl;
  18.         return 1;
  19.     }
  20.  
  21.     // 2. 要签名的原始数据（任意长度）
  22.     std::string message = "这是一段用于学习的原始测试数据，可以很长，可以包含中文。";
  23.  
  24.     std::cout << "原始数据: " << message << std::endl;
  25.     std::cout << "原始数据长度: " << message.size() << " 字节" << std::endl;
  26.  
  27.     // 3. 签名 - 直接传入原始数据
  28.     unsigned char signature[512];     // 缓冲区要足够大
  29.     unsigned int sig_len = 0;
  30.  
  31.     int ret = ECDSA_sign(
  32.         0,                                      // nid 类型（0 表示默认）
  33.         (const unsigned char*)message.data(),   // ← 原始数据指针
  34.         (int)message.size(),                    // ← 原始数据长度

  36.         &sig_len,                               // 输出签名实际长度
  37.         eckey                                   // 私钥
  38.     );
  39.  
  40.     if (ret != 1) {
  41.         std::cerr << "ECDSA_sign 签名失败！" << std::endl;

  43.         return 1;
  44.     }
  45.  
  46.     std::cout << "签名成功！签名长度 = " << sig_len << " 字节" << std::endl;

  48.  
  49.     // 4. 验证 - 同样直接传入原始数据
  50.     int verify_ret = ECDSA_verify(
  51.         0,                                      // nid 类型
  52.         (const unsigned char*)message.data(),   // ← 必须传入和签名时完全相同的原始数据
  53.         (int)message.size(),                    // ← 必须是相同的长度
  54.         signature,                              // 签名数据
  55.         sig_len,                                // 签名长度
  56.         eckey                                   // 公钥（同一个密钥对）
  57.     );
  58.  
  59.     if (verify_ret == 1)
  60.         std::cout << "ECDSA_verify 验证成功！" << std::endl;
  61.     else if (verify_ret == 0)
  62.         std::cout << "验证失败！" << std::endl;
  63.     else
  64.         std::cout << "验证出错！" << std::endl;
  65.  
  66.     EC_KEY_free(eckey);
  67.     return 0;
  68. }
