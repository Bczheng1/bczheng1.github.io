2.AcceptBlock

CBlock里有个函数，AcceptBlock,当生成块时，这个块的nonce已经被找出来了，就会开始调用AcceptBlock函数，将区块存储起来（包括写到磁盘文件里）。

CBlock 类 中AccetpBlock函数代码如下：

  1. bool CBlock::AcceptBlock()
  2. {
  3.     // 检查是否是重复区块（duplicate）
  4.     // 计算当前区块的哈希值（hash）
  5.     uint256 hash = GetHash();
  6.     
  7.     // mapBlockIndex 是一个全局的 map，key 是区块哈希，value 是 CBlockIndex*（区块在链中的索引信息）
  8.     // 如果这个哈希已经存在于 mapBlockIndex 中，说明这个区块我们之前已经处理过了
  9.     if (mapBlockIndex.count(hash))
  10.         return error("AcceptBlock() : block already in mapBlockIndex");  // 返回错误并记录日志
  11.  
  12.     // 获取前一个区块（prev block）的索引
  13.     // hashPrevBlock 是当前区块头中记录的上一个区块的哈希
  14.     map<uint256, CBlockIndex*>::iterator mi = mapBlockIndex.find(hashPrevBlock);
  15.     
  16.     // 如果找不到前一个区块，说明这个区块是“孤块”（orphan block），无法连接到现有链上
  17.     if (mi == mapBlockIndex.end())
  18.         return error("AcceptBlock() : prev block not found");
  19.     
  20.     // pindexPrev 指向前一个区块的索引对象，后面会用来做各种检查
  21.     CBlockIndex* pindexPrev = (*mi).second;
  22.  
  23.     // 检查时间戳（timestamp）是否合法
  24.     // nTime 是当前区块的时间戳
  25.     // GetMedianTimePast() 是前一个区块及其前面若干区块的中位时间（防止时间戳被恶意提前）
  26.     // 要求：当前区块时间必须 > 前一个区块的中位时间
  27.     if (nTime <= pindexPrev->GetMedianTimePast())
  28.         return error("AcceptBlock() : block's timestamp is too early");
  29.

  32.     // GetNextWorkRequired() 根据前一个区块计算出当前区块应该使用的难度
  33.     // 如果不匹配，说明矿工没有正确完成挖矿工作
  34.     if (nBits != GetNextWorkRequired(pindexPrev))
  35.         return error("AcceptBlock() : incorrect proof of work");
  36.  
  37.     // 把区块实际写入磁盘（block history file）
  38.     // !fClient 表示不是轻客户端模式（完整节点需要保存完整区块数据）

  40.     unsigned int nFile;
  41.     unsigned int nBlockPos;
  42.     if (!WriteToDisk(!fClient, nFile, nBlockPos))
  43.         return error("AcceptBlock() : WriteToDisk failed");  // 写入失败就报错
  44.  
  45.     // 把这个区块加入到区块索引（mapBlockIndex 和链结构）中

  47.     if (!AddToBlockIndex(nFile, nBlockPos))

  51.     // 则向网络中其他节点广播这个区块（RelayInventory）
  52.     if (hashBestChain == hash)
  53.         RelayInventory(CInv(MSG_BLOCK, hash));
  54.  
  55.     // （已注释掉的旧代码）
  56.     // 早期版本曾经有一个“原子”（atoms）机制，用于给新产生的币添加一些用户评论或标记
  57.     // 后来被移除，不再使用

3.WriteToDisk

我们来看一下这个函数据代码：

  1. // 把当前区块写入磁盘的历史区块文件（blk?????.dat）中
  2. // fWriteTransactions = true 表示写入完整区块（包含所有交易）
  3. //                     = false 表示只写入区块头（轻客户端/headers-only 模式使用）
  4. // nFileRet     : 输出参数，返回写入的文件编号（例如 0 表示 blk00000.dat）
  5. // nBlockPosRet : 输出参数，返回该区块在文件中的起始偏移位置（字节位置）
  6. bool WriteToDisk(bool fWriteTransactions, unsigned int& nFileRet, unsigned int& nBlockPosRet)
  7. {
  8.     // 1. 打开（或创建）历史区块文件，用于追加写入
  9.     // AppendBlockFile 会自动管理 blk00000.dat、blk00001.dat 等文件
  10.     // 当当前文件太大时，会自动切换到下一个文件
  11.     // nFileRet 会返回当前使用的文件编号
  12.     CAutoFile fileout = AppendBlockFile(nFileRet);
  13.     
  14.     // 如果打开文件失败（磁盘满、权限问题等），返回错误
  15.     if (!fileout)
  16.         return error("CBlock::WriteToDisk() : AppendBlockFile failed");
  17.  
  18.     // 如果不需要写入交易（只想存区块头），则设置序列化标志
  19.     // SER_BLOCKHEADERONLY 是一个特殊的序列化模式，只会序列化区块头，不序列化交易列表
  20.     if (!fWriteTransactions)
  21.         fileout.nType |= SER_BLOCKHEADERONLY;
  22.  
  23.     // 2. 写入索引头部（Index Header）
  24.     // 先计算整个区块（或区块头）序列化后的大小
  25.     unsigned int nSize = fileout.GetSerializeSize(*this);
  26.     
  27.     // 写入固定的消息起始标识（pchMessageStart，通常是 0xF9 0xBE 0xB4 0xD9）
  28.     // 再写入区块数据的大小 nSize
  29.     // 这样后面读取文件时，可以快速跳过或定位每个区块
  30.     fileout << FLATDATA(pchMessageStart) << nSize;
  31.  
  32.     // 3. 记录当前区块在文件中的起始位置
  33.     // ftell() 获取文件当前指针的位置（即区块数据的开始位置）
  34.     nBlockPosRet = ftell(fileout);
  35.     if (nBlockPosRet == -1)
  36.         return error("CBlock::WriteToDisk() : ftell failed");
  37.  
  38.     // 4. 真正把区块数据写入文件
  39.     // 这里会调用 CBlock 的序列化函数 operator<< 
  40.     // 根据前面设置的 SER_BLOCKHEADERONLY 标志，决定是写完整区块还是只写区块头
  41.     fileout << *this;
  42.  
  43.     // 所有步骤都成功
  44.     return true;
  45. }

4.CAutoFile

可以看到是调用了CAutoFile类来写区块数据到磁盘的,这个CAutoFile,只是对系统文件操作函数，如fopen,fread,fwrite进行了一些封装。

最终fileout<<*this，这里的*this就是CBlock类型，将区块写进文件中。注意这里是直接写二进制数据的，而且几乎没什么特殊处理的，是直接这样写的。然后下一个区块，也是这样追加写进去的。

文件名blk00001.dat，这样写，大概写满（2GB），又创建下一个blk00002.dat这样接着写。

但是现在问题来了，这个写区块数据，可以说是交易数据，没经过DB数据库。

那么之前我们看到了db.cpp里，有tx标识的数据，就是CTxDB类相关的数据，是什么呢？

这个也是跟交易数据相关，只不过这个是交易数据的索引目录。具体交易数据是存在blk00001.dat是二进制流这样的文件中的。

为什么要分开写，交易数据其实很大的，用DB键值对写，明显不现实。所以我们直接写二进制流到文件中。不经过数据库。

但是这种数据我们如果要读交易数据，需要大量遍历还原查找，就会非常的不方便。

那么一个区块哈希我们保存它对应的文件名和偏移位置，就能直接读取。

所以这个索引数据，就保存在CTxDB对应的数据库里，对应名是：blkindex.dat

6.AddToBlockIndex

块数据写入文件名后，我们就得了文件名nFile（编号），和这个块在文件中的偏移位置nBlockPos，就可以调用AddToBlockIndex(nFile, nBlockPos)，构建索引了，这里面就是写入CTxDB数据了，也就是之前说的DB数据库里的交易部分数据（其实是交易索引数据)

  1. // 把一个已经写入磁盘的区块，正式加入到内存中的区块索引系统中
  2. // 并可能更新当前最优链（best chain）
  3. bool CBlock::AddToBlockIndex(unsigned int nFile, unsigned int nBlockPos)
  4. {
  5.     // 检查是否重复（double-check）
  6.     // 计算当前区块的哈希
  7.     uint256 hash = GetHash();
  8.     
  9.     // 如果这个区块哈希已经存在于 mapBlockIndex 中，说明重复
  10.     if (mapBlockIndex.count(hash))
  11.         return error("AddToBlockIndex() : %s already exists", hash.ToString().substr(0,14).c_str());
  12.  
  13.     // 构造一个新的区块索引对象 CBlockIndex
  14.     // 把文件编号、文件内偏移位置、以及当前区块的信息传进去
  15.     CBlockIndex* pindexNew = new CBlockIndex(nFile, nBlockPos, *this);
  16.     if (!pindexNew)
  17.         return error("AddToBlockIndex() : new CBlockIndex failed");
  18.  
  19.     // 将新创建的 CBlockIndex 插入到全局的 mapBlockIndex 中
  20.     // mapBlockIndex: key = 区块哈希, value = CBlockIndex*
  21.     map<uint256, CBlockIndex*>::iterator mi = 
  22.         mapBlockIndex.insert(make_pair(hash, pindexNew)).first;
  23.  
  24.     // 让 pindexNew 的 phashBlock 指针指向 map 中实际存储的哈希（避免拷贝）
  25.     pindexNew->phashBlock = &((*mi).first);
  26.  
  27.     // 查找前一个区块的索引
  28.     map<uint256, CBlockIndex*>::iterator miPrev = mapBlockIndex.find(hashPrevBlock);
  29.     
  30.     // 如果找到了前一个区块，则建立父子关系，并计算当前区块的高度
  31.     if (miPrev != mapBlockIndex.end())
  32.     {
  33.         pindexNew->pprev = (*miPrev).second;           // 指向前一个区块索引
  34.         pindexNew->nHeight = pindexNew->pprev->nHeight + 1;  // 高度 = 前一个高度 + 1
  35.     }
  36.  
  37.     // 打开交易数据库（Berkeley DB），用于持久化区块索引信息
  38.     CTxDB txdb;
  39.     txdb.TxnBegin();                                   // 开始一个数据库事务
  40.  
  41.     // 把区块索引信息写入磁盘数据库（leveldb/BerkeleyDB）
  42.     // CDiskBlockIndex 是用于磁盘存储的精简版 CBlockIndex
  43.     txdb.WriteBlockIndex(CDiskBlockIndex(pindexNew));
  44.  
  45.     // ==================== 判断是否成为新的最优链（New best） ====================
  46.  
  47.     // 如果当前区块的高度 大于 当前记录的最优链高度（nBestHeight）
  48.     if (pindexNew->nHeight > nBestHeight)
  49.     {
  50.         // 特殊情况：这是创世区块（Genesis Block）
  51.         if (pindexGenesisBlock == NULL && hash == hashGenesisBlock)
  52.         {
  53.             pindexGenesisBlock = pindexNew;                // 记录创世区块索引
  54.             txdb.WriteHashBestChain(hash);                 // 把最优链哈希写入数据库
  55.         }
  56.         // 正常情况：这个区块直接接在当前最优链的末尾（连续扩展）
  57.         else if (hashPrevBlock == hashBestChain)
  58.         {
  59.             // 添加到当前最优分支
  60.             if (!ConnectBlock(txdb, pindexNew) || !txdb.WriteHashBestChain(hash))
  61.             {
  62.                 // 如果连接区块失败（交易验证失败等），则回滚
  63.                 txdb.TxnAbort();
  64.                 pindexNew->EraseBlockFromDisk();           // 删除已写入磁盘的区块数据
  65.                 mapBlockIndex.erase(pindexNew->GetBlockHash()); // 从内存索引中删除
  66.                 delete pindexNew;                          // 释放内存
  67.                 return error("AddToBlockIndex() : ConnectBlock failed");
  68.             }
  69.  
  70.             txdb.TxnCommit();                              // 提交数据库事务
  71.  
  72.             // 更新链表指针：让前一个区块的 pnext 指向当前新区块
  73.             pindexNew->pprev->pnext = pindexNew;
  74.  
  75.             // 从内存交易池（mempool）中删除已经打包进这个区块的交易
  76.             foreach(CTransaction& tx, vtx)
  77.                 tx.RemoveFromMemoryPool();
  78.         }
  79.         // 其他情况：这个区块属于一条新的分支，需要进行链重组（reorg）
  80.         else
  81.         {
  82.             // 执行链重组：可能需要切换到这条更优的分支
  83.             if (!Reorganize(txdb, pindexNew))
  84.             {
  85.                 txdb.TxnAbort();
  86.                 return error("AddToBlockIndex() : Reorganize failed");
  87.             }
  88.         }
  89.  
  90.         // 更新全局最优链相关变量
  91.         hashBestChain = hash;          // 当前最优链的末尾哈希
  92.         pindexBest = pindexNew;        // 当前最优链的末尾区块索引
  93.         nBestHeight = pindexBest->nHeight;   // 更新最优链高度
  94.         nTransactionsUpdated++;        // 通知钱包等模块交易已更新
  95.  
  96.         // 打印日志：新最优区块
  97.         printf("AddToBlockIndex: new best=%s  height=%d\n", 
  98.                hashBestChain.ToString().substr(0,14).c_str(), nBestHeight);
  99.     }
  100.  
  101.     // 如果不是新最优链（只是普通分支），也需要提交数据库事务
  102.     txdb.TxnCommit();
  103.     txdb.Close();                      // 关闭数据库连接
  104.  
  105.     // 如果这个区块成为了新的最优区块，则把钱包中还未上链的交易重新广播
  106.     if (pindexNew == pindexBest)
  107.         RelayWalletTransactions();
  108.  
  109.     // 通知界面刷新（早期比特币有 GUI）
  110.     MainFrameRepaint();
  111.  
  112.     return true;   // 成功加入区块索引
  113. }

7.CBlockIndex

我们可以看到上面的函数，大量的用到了CBlockIndex这个类型，这就是区块索引类，主要存储一个区块在二进制文件中的位置，同时也包含一些该区块头的数据，nVersion,nBits等，还有该索引，在mapBlockIndex（第二参数类型）中下一个CBLockIndex *pnext;和上一个pprev;以及自身的phashBlock,哈希值，即mapBlockIndex中的第一个参数，key。

这样的构建有什么好处的呢，即，你如果知道其中一个索引（通过区块哈希值），你就可以根据pprev;pnext;遍历整个区块索引，也可以读取整个区块数据。那么永远如果存储最新（最后）的一个索引，就可以随时读取整个区块数据了(并构建)。

  1. // 区块链是一个树状结构（tree shaped structure）
  2. // 以创世区块（genesis block）作为根节点
  3. // 每个区块都可能有多个候选区块作为它的下一个区块（因为挖矿存在分叉）
  4. // pprev 和 pnext 用于链接当前最长主链（main/longest chain）的一条路径
  5. // 一个区块可能有多个子区块指向它（多个 pprev 指向同一个区块），但 pnext 只指向最长链的方向
  6. // 如果该区块不在最长链上，则 pnext 通常为 NULL
  7. //
  8. class CBlockIndex
  9. {
  10. public:

  130.                 return nTime;            // 返回当前区块自己的时间
  131.             pindex = pindex->pnext;      // 向后移动
  132.         }
  133.  
  134.         // 如果能向前看够 5 个区块，则返回那个位置的 MedianTimePast
  135.         return pindex->GetMedianTimePast();
  136.     }

  6. {
  7.     nFileRet = 0;                    // 先把输出参数清零，表示初始状态“没有找到可用文件”
  8.  
  9.     loop                             // 这是 Bitcoin 早期代码中定义的宏，等同于 while(true) 无限循环

  24.         // FAT32 理论最大 4GB，但 fseek/ftell 在很多系统上只能正确处理 2GB 以内的文件
  25.         // 所以 Satoshi 设置了一个安全阈值：文件大小必须小于 2GB - MAX_SIZE（留一点余量）
  26.         if (ftell(file) < 0x7F000000 - MAX_SIZE)
  27.         {

  34.         fclose(file);                // 关闭当前文件

10.OpenBlockFile

AppendBlockFile调用了OpenBlockFile函数，将此函数代码复制到main.cpp中。

  1. // 打开比特币区块数据文件（blkXXXXX.dat）
  2. // 参数说明：
  3. //   nFile     : 文件编号，例如 1 表示 blk0001.dat，2 表示 blk0002.dat
  4. //   nBlockPos : 在文件中的偏移位置（字节）。如果为 0，则打开后指针在文件开头/结尾（取决于模式）
  5. //   pszMode   : 文件打开模式，例如 "rb"、"ab"、"r+b" 等（和 fopen 的 mode 参数完全一致）
  6. FILE* OpenBlockFile(unsigned int nFile, unsigned int nBlockPos, const char* pszMode)
  7. {
  8.     // 如果传入的文件编号是 -1（无效值），直接返回失败
  9.     // 这是一种简单的错误防护
  10.     if (nFile == -1)
  11.         return NULL;
  12.  
  13.     // 使用 strprintf 构造完整的文件路径和文件名
  14.     // 例如：在 Windows 下会生成类似 "C:\Users\xxx\AppData\Roaming\Bitcoin\blk0001.dat"
  15.     // GetAppDir() 返回比特币的数据目录（通常是 %APPDATA%\Bitcoin 或 ~/.bitcoin）
  16.     FILE* file = fopen(
  17.                      strprintf("%s\\blk%04d.dat", GetAppDir().c_str(), nFile).c_str(), 
  18.                      pszMode
  19.                  );
  20.  
  21.     // 如果 fopen 失败（文件不存在、权限不足、磁盘满等），直接返回 NULL
  22.     if (!file)
  23.         return NULL;
  24.  
  25.     // ==================== 重要逻辑：是否需要定位到指定位置 ====================
  26.     // 只有满足以下三个条件，才会执行 fseek 移动文件指针：
  27.     // 1. nBlockPos 不等于 0（即调用者明确指定了要跳到某个位置）
  28.     // 2. 当前打开模式中**不包含** 'a'（append 追加模式）
  29.     // 3. 当前打开模式中**不包含** 'w'（write 从头写入模式）
  30.     //
  31.     // 为什么这样判断？
  32.     // - "ab"、"a"、"a+" 等追加模式下，文件指针自动在末尾，fseek 通常无效或无意义
  33.     // - "w"、"w+" 等模式会清空文件，从头开始，也不应该随意 fseek
  34.     // - 只有读模式（如 "rb"、"r+b"）或需要精确写入指定位置时，才需要手动定位
  35.     if (nBlockPos != 0 && !strchr(pszMode, 'a') && !strchr(pszMode, 'w'))
  36.     {
  37.         // 把文件指针移动到文件内指定的字节偏移位置（从文件开头算起）
  38.         if (fseek(file, nBlockPos, SEEK_SET) != 0)
  39.         {
  40.             // 如果 fseek 失败（比如位置超出文件大小、文件被锁等）
  41.             fclose(file);      // 先关闭已经打开的文件，避免资源泄漏
  42.             return NULL;       // 返回失败
  43.         }
  44.     }
  45.  
  46.     // 一切正常，返回打开的文件指针
  47.     return file;
  48. }
